def consultarCNPJ(request): cnpj = request.GET.get('cnpj', '').replace('.', '').replace('/', '').replace('-', '') dados = None erro = None if cnpj: try: response = requests.get( f'https://brasilapi.com.br/api/cnpj/v1/{cnpj}', timeout=10 ) if response.status_code == 200: dados = response.json() elif response.status_code == 429: erro = 'Muitas consultas realizadas. Aguarde um pouco e tente novamente.' elif response.status_code == 404: erro = 'CNPJ não encontrado.' elif response.status_code == 400: erro = 'CNPJ inválido.' else: erro = f'Erro ao consultar CNPJ. Código: {response.status_code}' except requests.exceptions.RequestException: erro = 'Não foi possível conectar à API.' return render(request, 'consultaCNPJ.html', { 'dados': dados, 'erro': erro })
